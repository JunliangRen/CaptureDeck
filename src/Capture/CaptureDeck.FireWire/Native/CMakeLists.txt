cmake_minimum_required(VERSION 3.25)
project(CaptureDeckFireWireProbe LANGUAGES CXX)

if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "The FireWire probe requires Windows and MSVC")
endif()
if(NOT CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "x64")
    message(FATAL_ERROR "Configure an x64 build")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /permissive- /utf-8 /EHsc)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)

add_library(capdvhs_core STATIC src/ts.cpp src/util.cpp src/pipeline.cpp)
target_include_directories(capdvhs_core PUBLIC include)
target_link_libraries(capdvhs_core PUBLIC ole32)
add_library(capdvhs_dshow STATIC src/device.cpp src/sink.cpp)
target_include_directories(capdvhs_dshow PUBLIC include)
target_link_libraries(capdvhs_dshow PUBLIC capdvhs_core strmiids ole32 oleaut32 uuid setupapi cfgmgr32 advapi32)
add_executable(CaptureDeck.FireWire.Probe src/main.cpp src/capture.cpp)
target_link_libraries(CaptureDeck.FireWire.Probe PRIVATE capdvhs_dshow)

include(CTest)
if(BUILD_TESTING)
    set(test_sources "${CMAKE_CURRENT_SOURCE_DIR}/../../../../UT/CaptureDeck.FireWire.UT")
    foreach(name ts device sink pipeline)
        add_executable(${name}_tests "${test_sources}/${name}_tests.cpp")
        target_link_libraries(${name}_tests PRIVATE capdvhs_dshow)
        add_test(NAME ${name} COMMAND ${name}_tests)
    endforeach()
    add_test(NAME cli_help COMMAND CaptureDeck.FireWire.Probe --help)
    add_test(NAME cli_version COMMAND CaptureDeck.FireWire.Probe --version)
    add_executable(capture_tests "${test_sources}/capture_tests.cpp" src/capture.cpp)
    target_link_libraries(capture_tests PRIVATE capdvhs_dshow)
    add_test(NAME capture_integration COMMAND capture_tests "${CMAKE_CURRENT_BINARY_DIR}/test-data/capture")
    set_tests_properties(capture_integration PROPERTIES TIMEOUT 30)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME cli_integration COMMAND "${Python3_EXECUTABLE}"
                 "${test_sources}/cli_tests.py"
                 "$<TARGET_FILE:CaptureDeck.FireWire.Probe>" "${CMAKE_CURRENT_BINARY_DIR}/test-data/cli")
    endif()
endif()

install(TARGETS CaptureDeck.FireWire.Probe RUNTIME DESTINATION .)
